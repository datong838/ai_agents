### 3.3 重要度计算

```python
def calculate_importance(episode: Episode, task_result: TaskResult) -> float:
    """计算 Episode 重要度，决定 TTL 和提升。

    四因子加权（参考认知科学情景记忆强度模型）：
    """
    factors = {}

    # 1. 成败因子：失败教训 > 成功经验（0.3 权重）
    factors["outcome"] = 0.8 if not task_result.success else 0.5

    # 2. 复用因子：可复用平台数越多越重要（0.25 权重）
    reuse_count = len(episode.reusable_for_platforms)
    factors["reusability"] = min(1.0, 0.3 + reuse_count * 0.2)

    # 3. 稀缺因子：该任务类型历史 Episode 越少越重要（0.25 权重）
    history_count = await episodic.count_by_type(episode.task_type, episode.platform)
    factors["scarcity"] = max(0.2, 1.0 - history_count * 0.1)

    # 4. 反馈因子：上游反馈有用（0.2 权重）
    factors["feedback"] = 0.5  # 默认，待上游反馈更新

    importance = (
        factors["outcome"] * 0.3 +
        factors["reusability"] * 0.25 +
        factors["scarcity"] * 0.25 +
        factors["feedback"] * 0.2
    )
    episode.importance_factors = factors
    return round(importance, 3)
```

```python
        Args:

        auth_type: str,
```

### 4.1 WorkingMemory 数据模型

```python
# aip_working_memory.py（新增）

class WorkingMemoryEntry(BaseModel):
    """Working 层条目 — 当前任务上下文。"""
    key: str                    # 如 "step_3_output"
    value: Any                  # 实际值 或 指针 "[POINTER:ref_id]"
    is_pointer: bool = False
    token_estimate: int = 0
    created_at: float = Field(default_factory=lambda: time.time())
    ref_id: str | None = None   # 指针引用 ID


class WorkingMemory:
    """Working 层 — 当前任务的工作记忆。

    参考 Claude Code 指针策略：超出 token 限制时用指针代替内容。
    参考 Claude Blog Progressive Disclosure：按步骤加载，不预加载。
    """

    MAX_CONTEXT_TOKENS = 8000

    def __init__(self, task_id: str, external_store: ExternalStore):
        self._task_id = task_id
        self._external = external_store  # Redis（持久化检查点）
        self._context: dict[str, WorkingMemoryEntry] = {}
        self._token_count = 0
```

### 4.2 指针策略（扩展 00-总览）

```python
    def set(self, key: str, value: Any) -> None:
        """设置上下文变量。超出限制自动转指针。"""
        token_estimate = self._estimate_tokens(value)

        if self._token_count + token_estimate > self.MAX_CONTEXT_TOKENS:
            # 转指针：存到 Redis，只保留引用
            ref_id = f"wm:{self._task_id}:{key}:{uuid.uuid4().hex[:6]}"
            await self._external.setex(ref_id, 3600, json.dumps(value))  # 1h TTL

            self._context[key] = WorkingMemoryEntry(
                key=key,
                value=f"[POINTER: {ref_id}]",
                is_pointer=True,
                token_estimate=20,  # 指针固定 20 token
                ref_id=ref_id,
            )
            self._token_count += 20
        else:
            self._context[key] = WorkingMemoryEntry(
                key=key,
                value=value,
                token_estimate=token_estimate,
            )
            self._token_count += token_estimate

    async def get(self, key: str) -> Any:
        """获取值。指针按需加载。"""
        entry = self._context.get(key)
        if entry is None:
            return None
        if entry.is_pointer and entry.ref_id:
            raw = await self._external.get(entry.ref_id)
            return json.loads(raw) if raw else None
        return entry.value
```

### 4.3 八段式压缩算法

参考 Claude Code 的上下文压缩：

```python
    async def compress(self) -> None:
        """上下文压缩 — 八段式。

        策略：
        1. 保留最近 3 步的完整上下文（Think/Act/Verify/Observe）
        2. 第 4-6 步压缩为摘要（LLM 生成 ≤ 200 token）
        3. 第 7 步以前转为指针（只保留 ref_id）
        """
        steps = sorted(
            [k for k in self._context if k.startswith("step_")],
            key=lambda k: int(k.split("_")[1])
        )

        for i, step_key in enumerate(steps):
            entry = self._context[step_key]
            step_num = i + 1
            total = len(steps)

            if step_num > total - 3:
                # 最近 3 步：保留完整
                continue
            elif step_num > total - 6:
                # 4-6 步前：压缩为摘要
                if not entry.is_pointer:
                    summary = await self._llm_summarize(entry.value, max_tokens=200)
                    entry.value = summary
                    entry.token_estimate = self._estimate_tokens(summary)
                    self._token_count = sum(e.token_estimate for e in self._context.values())
            else:
                # 更早：转指针
                if not entry.is_pointer:
                    await self._to_pointer(step_key, entry)
```

### 4.4 渐进式披露（与 00-总览 §九对齐）

```python
    async def load_for_step(self, task: Task, step: TaskStep) -> dict:
        """按当前步骤需要，渐进式加载记忆。

        与 00-总览 §9.2 一致：
        - Working 层始终加载
        - Semantic 层按步骤类型按需检索
        - Episodic 层按相似度按需检索
        """
        context = {"working": self.get_all()}

        if step.action_type in ("llm_call", "branch"):
            context["semantic"] = await self._semantic.retrieve(
                query=f"{task.type} {step.action_type}",
                top_k=3,
                filters=RetrievalFilters(owner_agents=[task.agent]),
            )

        if step.action_type in ("llm_call", "action_writeback"):
            context["episodic"] = await self._episodic.retrieve(
                EpisodicQuery(task_type=task.type, platform=task.context.get("platform")),
                top_k=2,
            )

        return context
```

## 五、存储选型决策

> 规划文档 §六标记 "Q3 行业 Wiki 向量存储选型" 为 P1 阻塞问题。本节给出决策。

### 5.1 向量库三选一

| 维度 | Milvus | Qdrant | pgvector |
|------|--------|--------|----------|
| 部署复杂度 | 中（需独立集群） | 低（单容器） | **低**（PG 扩展） |
| 检索性能（100万向量） | 优 | 优 | 中 |
| 混合检索（向量+元数据过滤） | 强 | **强** | 中 |
| 与现有技术栈契合 | 需新增 | 需新增 | **复用 PostgreSQL** |
| 运维成本 | 高 | 中 | **低** |
| 电商知识库规模预估 | <50万条 | <50万条 | <50万条 |

**决策：pgvector**

**理由**：
1. 美妆行业知识库冷启动后规模预估 <50万条，pgvector 性能足够
2. 现有 `ontology_wiki_engine.py` 已用 PostgreSQL，向量与结构化数据同库，避免双写一致性问题
3. 运维成本最低（不新增独立集群）
4. 若未来规模超百万，可平滑迁移到 Qdrant（接口抽象，最小更改）

## 六、检索策略 — 混合检索 + 重排

### 6.1 混合检索流程

```python
async def hybrid_retrieve(
    query: str,
    top_k: int = 5,
    filters: RetrievalFilters | None = None,
) -> list[KnowledgeEntry]:
    """混合检索：向量 + 关键词 + 元数据过滤。

    1. 向量检索（语义相似）— 召回 3×top_k
    2. 关键词检索（BM25/PG全文）— 召回 2×top_k
    3. 合并去重
    4. 元数据过滤（category/tags/confidence/platform_scope）
    5. 重排（RRF 融合 + 时效衰减）
    6. 取 top_k
    """
    # 1. 向量召回
    query_emb = await embed(query)
    vector_hits = await pgvector_search(query_emb, limit=top_k * 3, filters=filters)

    # 2. 关键词召回
    keyword_hits = await pg_fulltext_search(query, limit=top_k * 2, filters=filters)

    # 3. 合并去重
    candidates = dedup_by_id(vector_hits + keyword_hits)

    # 4. 重排
    reranked = rrf_rerank(candidates, vector_weight=0.6, keyword_weight=0.4)

    # 5. 时效衰减（近期知识加权）
    reranked = apply_recency_boost(reranked, half_life_days=30)

    # 6. 可信度加权（L1/L2 > L3/L4）
    reranked = apply_confidence_weight(reranked)

    return reranked[:top_k]
```

### 6.2 RRF 融合公式

```
score(d) = Σ (1 / (k + rank_i(d)))

k = 60（标准 RRF 参数）
rank_i(d) = 文档 d 在第 i 个检索器中的排名
```

### 6.3 检索过滤策略

```python
class RetrievalFilters(BaseModel):
    """检索过滤条件。"""
    category: list[str] | None = None        # 限定类别
    tags: list[str] | None = None            # 限定标签
    confidence_level: list[str] | None = None  # 限定可信度，如 ["L1", "L2"]
    platform_scope: list[str] | None = None    # 限定平台
    owner_agents: list[str] | None = None      # 限定可用数字同事
    effective_only: bool = True                 # 只返回有效期内
```

## 七、提升机制 — Working → Episodic → Semantic

### 7.1 三级提升触发条件

```
Working Memory（任务进行中）
    │
    │ 触发：任务结束
    │ 条件：有可复用经验（reusability_score > 0.5）或失败教训
    ▼
Episodic Memory（90 天 TTL）
    │
    │ 触发：该模式出现 ≥ 5 次（auto_pattern_detection）
    │ 条件：importance ≥ 0.7 且 positive_feedback > negative_feedback
    ▼
Semantic Memory（长期）
    │
    │ 触发：人工审核通过（L3/L4 需 Draft 审批）
    │ 条件：confidence_level 从 L3 提升到 L2
```

### 7.2 提升实现

```python
class MemoryPromoter:
    """记忆提升器。"""

    async def promote_working_to_episodic(self, task: Task) -> str | None:
        """Working → Episodic（任务结束时）。"""
        important = await self._extract_important_context(task)
        if not important:
            return None

        episode = Episode(
            task_type=task.type,
            agent=task.agent,
            platform=task.context.get("platform", ""),
            summary=important["summary"],
            success=task.success,
            key_decisions=important["decisions"],
            lessons_learned=important["lessons"],
            importance=calculate_importance(episode, task.result),
        )
        return await self._episodic.store(episode)

    async def promote_episodic_to_semantic(self, pattern: DetectedPattern) -> str | None:
        """Episodic → Semantic（高频模式自动检测后）。"""
        if pattern.frequency < 5:
            return None

        entry = KnowledgeEntry(
            title=f"模式: {pattern.name}",
            content=pattern.description,
            category=pattern.category,
            confidence_level="L3",  # 经验级，需治理审核
            source="auto_pattern_detection",
            owner_agents=pattern.agents,
        )
        # 走 03-治理写入门控
        return await self._governance.write(entry)
```

### 7.3 自动模式检测

```python
class PatternDetector:
    """自动检测高频模式（Episodic → Semantic 触发器）。"""

    async def detect_patterns(self, task_type: str, window_days: int = 30) -> list[DetectedPattern]:
        """检测近 30 天的高频模式。"""
        episodes = await self._episodic.find_by_type(task_type, days=window_days)

        # 聚类相似 Episode
        clusters = await self._cluster_episodes(episodes)

        patterns = []
        for cluster in clusters:
            if len(cluster) >= 5:  # 5 次以上
                patterns.append(DetectedPattern(
                    name=self._name_cluster(cluster),
                    description=self._summarize_cluster(cluster),
                    frequency=len(cluster),
                    category=self._infer_category(cluster),
                    agents=list(set(e.agent for e in cluster)),
                ))

        return patterns
```

```python
# 上游（TAOR 循环 / FDE 编排器 / Evals）唯一入口
class MemorySystem:
    """三层记忆统一门面。"""

    def __init__(self, semantic: SemanticLayer, episodic: EpisodicLayer, working_factory: WorkingMemoryFactory):
        self._semantic = semantic
        self._episodic = episodic
        self._working_factory = working_factory

    # ── TAOR 循环调用 ──
    async def load_for_step(self, task: Task, step: TaskStep) -> dict:
        """渐进式加载（见 §4.4）。"""

    def get_working(self, task_id: str) -> WorkingMemory:
        """获取任务的 Working Memory。"""

    # ── 任务结束时调用 ──
    async def on_task_complete(self, task: Task, result: TaskResult) -> None:
        """任务完成：Working→Episodic 提升 + 管道2自学习。"""
        await self._promoter.promote_working_to_episodic(task)
        if result.success:
            await self._pipeline.pipe2_self_learning(result)

    # ── FDE 跨平台复用 ──
    async def retrieve_fde_experience(
        self, platform: str, auth_type: str, skill_id: str
    ) -> list[Episode]:
        """FDE 三级检索（同平台→同认证→通用）。"""
        return await self._episodic.retrieve_fde_three_tier(platform, auth_type, skill_id)

    # ── Evals 门控 ──
    async def get_reflection_rules(self, agent: str) -> list[KnowledgeEntry]:
        """获取数字同事的 Reflection 规则（供 Evals 注册）。"""
        return await self._semantic.retrieve(
            query="reflection rules",
            filters=RetrievalFilters(category=["expert_knowledge"], owner_agents=[agent]),
            top_k=50,
        )
```

## 十一、关键设计决策

| # | 决策 | Why | 影响 |
|---|------|-----|------|
| 1 | **pgvector** 而非 Milvus/Qdrant | 美妆知识库 <50 万条，pgvector 足够；复用 PostgreSQL，运维成本最低；接口抽象可平滑迁移 | 不新增独立向量集群 |
| 2 | **混合检索**（向量+关键词+RRF）而非纯向量 | 电商术语精确匹配（SKU/成分名）关键词更准；语义模糊查询向量更准；两者融合召回率最高 | 检索延迟增加 ~20ms（可接受） |
| 3 | **FDE 三级检索**（同平台→同认证→通用）与 13-FDE/10 §8.5 一致 | 跨平台复用是 FDE 核心价值；三级降级保证总有经验可用 | 第二次接入同类平台耗时降低 ≥50% |
| 4 | **八段式压缩**（最近3步完整/4-6步摘要/更早转指针） | 参考 Claude Code；保留近期上下文质量，压缩历史，控 token 成本 | Working Memory 平均 token 降低 60% |
| 5 | **提升机制三级**（Working→Episodic→Semantic）+ 频率 ≥5 次自动触发 | 避免低频噪声进 Semantic；高频模式才值得长期保存 | Semantic 层质量高，低冗余 |
| 6 | **重要度四因子**（outcome/reusability/scarcity/feedback） | 失败教训 > 成功经验；可复用性高的优先；稀缺经验保护；上游反馈校准 | Episodic 自动 TTL 和提升决策有量化依据 |
| 7 | **bge-large-zh-v1.5 本地部署** | 电商场景中文为主；私有化部署不调外部 API；1024 维兼顾效果和性能 | 零外部 API 依赖 |
| 8 | **MemorySystem 统一门面**，上游不直接访问各层 | 职责隔离；上游（TAOR/FDE/Evals）接口统一；底层存储变更不影响上游 | 上游代码零耦合存储细节 |
