# filter_orders.py — 订单清洗管道 · Filter 节点
# 变换仓库: order-cleanse-transforms  |  关联管道: 订单清洗管道
# 功能: 过滤已完成+金额有效的订单，过滤测试用户

from pyspark.sql import DataFrame, functions as F
from datetime import datetime, timedelta

TODAY = datetime(2026, 7, 21)

# ---- 变换入口 ----
def transform(df: DataFrame) -> DataFrame:
    """
    order-cleanse-transforms :: filter_orders
    Pipeline 节点类型: Filter
    输入: raw_orders (892,104 行)
    """
    filtered = (
        df
        # 1. 仅保留已完成/处理中订单
        .filter(
            F.col("status").isin(["completed", "processing"])
        )
        # 2. 过滤测试用户 (customer_id 以 TEST- 开头)
        .filter(
            ~F.col("customer_id").startswith("TEST-")
        )
        # 3. 金额校验 (> 0)
        .filter(F.col("amount") > 0)
        # 4. 时间窗口 (最近 90 天)
        .filter(
            F.col("created_at") >= F.lit(TODAY - timedelta(days=90))
        )
    )

    return filtered

# ---- 测试 (Pipeline 搭建前本地验证) ----
if __name__ == "__main__":
    from pyspark.sql import SparkSession
    spark = SparkSession.builder.getOrCreate()
    df = spark.read.parquet("/Raw/Ecom/orders")
    result = transform(df)
    print(f"Filtered: {result.count():,} rows")
