查询管理 （SQL 查询语句，从 ObjectSet / Dataset 拉取数据）
🔍
getCountries()
SELECT country, SUM(confirmed) FROM COVID_Cases GROUP BY country ORDER BY confirmed DESC
1.2s 返回 195 行
🔍
getDailyStats()
SELECT date, confirmed, deaths, recovered FROM COVID_Cases WHERE country = $selected_country
0.8s 参数绑定
🔍
getTopRegions()
SELECT region, confirmed FROM COVID_Cases WHERE country = $selected_country ORDER BY confirmed DESC LIMIT 10
0.5s 返回 10 行
🔍
getVaccineData()
SELECT country, vaccine_rate FROM Vaccine_Stats v JOIN COVID_Cases c ON v.country = c.country
3.8s ⚠️ JOIN 查询
🔍
getMonthlyTrend()
SELECT DATE_TRUNC('month', date) AS month, AVG(confirmed) FROM COVID_Cases GROUP BY 1 ORDER BY 1
2.1s 聚合查询
